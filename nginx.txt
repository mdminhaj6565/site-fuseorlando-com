# --- 4) Home page normalisation -> 301 ---
rewrite ^/index(\.html)?/?$ / permanent;

# --- 5) Pretty URLs ---
# 5a) /page.html -> /page (301) only when the file exists
if ($request_uri ~ ^/([^?]+)\.html($|\?)) { set $noext /$1; }
if (-f $document_root$request_uri) { set $noext_ok "${noext}"; }

# 5b) strip trailing slash (301)
rewrite ^/(.*)/$ /$1 permanent;

# 5c) extensionless -> .html (internal rewrite, stays 200)
try_files $uri $uri.html $uri/index.html =404;

# --- 6) Legacy extensions -> 301 only when the .html target exists ---
if ($is_args = '') {
	rewrite ^/([^.]*)\.(shtml|phtml|jhtml|htm|php|aspx|asp|jsp|apk)$ /$1 permanent;
}

# --- 7) Performance ---
gzip on;
gzip_comp_level 6;
gzip_min_length 256;
gzip_types text/plain text/css text/xml application/javascript application/json
           application/xml image/svg+xml application/rss+xml;

location ~* \.(css|js|jpg|jpeg|png|gif|svg|ico|woff2?)$ {
	expires 1y;
	add_header Cache-Control "public, immutable";
}
location ~* \.html$ {
	expires 0;
	add_header Cache-Control "public, max-age=0, must-revalidate";
}
